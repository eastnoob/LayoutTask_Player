# Protocol Scoring Generator Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a language-neutral protocol kit, JSON batch compiler, scoring reference pipeline, and object-level export columns for Layout Task Player.

**Architecture:** Keep the browser player static and simple. Add an offline authoring pipeline that validates canonical batch JSON, compiles it into current runtime `manifest.json + tasks/*.json`, emits private scoring references, and exports both relative and absolute object-level analysis rows. Rhino support is documentation/templates only in v1; tests use self-authored mock JSON fixtures.

**Tech Stack:** TypeScript, Zod, Vitest, Node `fs/promises`, existing `tsx` tooling, existing decoder/export utilities, standard JSON.

```ts
  task_id: z.string().min(1),
  title: z.string().optional(),
  display_image: displayImageSchema.optional(),

    asset: z.string().min(1),
    x: z.number().finite(),
    y: z.number().finite(),
    width: z.number().positive(),
    height: z.number().positive(),
  }),
  objects: z.array(batchObjectSchema).min(1),
  completion: completionSchema.optional(),
  recording: recordingSchema.optional(),
  output: outputSchema.optional(),
  data_save: dataSaveSchema.optional(),
  flow: flowSchema.optional(),
  stage: stageSchema.optional(),
  messages: messagesSchema.optional(),
  requirements: requirementsSchema.optional(),
  scoring: scoringSchema.optional(),
  metadata: metadataSchema.optional(),
});

export const batchSharedSchema = z.object({
  asset_library: z.string().min(1),
  background_library: z.string().min(1),
  behavior_library: z.string().min(1),
  world: worldSchema.optional(),
  stage: stageSchema.optional(),
  completion: completionSchema.optional(),
  recording: recordingSchema.optional(),
  output: outputSchema.optional(),
  data_save: dataSaveSchema.optional(),
  flow: flowSchema.optional(),
  messages: messagesSchema.optional(),
  requirements: requirementsSchema.optional(),
});

export const batchSchema = z
  .object({
    schema: z.literal("layouttask.batch.v1"),
    experiment_id: z.string().min(1),
    title: z.string().optional(),
    config_version: z.string().optional(),
    output_dir: z.string().min(1).optional(),
    shared: batchSharedSchema,
    trials: z.array(batchTrialSchema).min(1),
  })
  .superRefine((value, context) => {
    const taskIds = new Set<string>();
    for (const [trialIndex, trial] of value.trials.entries()) {
      if (taskIds.has(trial.task_id)) {
        context.addIssue({
          code: z.ZodIssueCode.custom,
          path: ["trials", trialIndex, "task_id"],
          message: `duplicate task_id: ${trial.task_id}`,
        });
      }
      taskIds.add(trial.task_id);

      const objectIds = new Set<string>();
      for (const [objectIndex, object] of trial.objects.entries()) {
        if (objectIds.has(object.id)) {
          context.addIssue({
            code: z.ZodIssueCode.custom,
            path: ["trials", trialIndex, "objects", objectIndex, "id"],
            message: `duplicate object id in trial ${trial.task_id}: ${object.id}`,
          });
        }
        objectIds.add(object.id);
      }

      if (!trial.world && !value.shared.world) {
        context.addIssue({
          code: z.ZodIssueCode.custom,
          path: ["trials", trialIndex, "world"],
          message: `trial ${trial.task_id} requires world or shared.world`,
        });
      }
    }
  });

export const scoringReferenceSchema = z.object({
  schema: z.literal("layouttask.scoring-reference.v1"),
  experiment_id: z.string().min(1),
  tasks: z.record(
    z.object({
      qid: z.string().min(1),
      metadata: metadataSchema.optional(),
      objects: z.record(
        z.object({
          role: z.enum(["fixed", "variable"]).optional(),
          group_id: z.string().min(1).optional(),
          target: objectTargetSchema.optional(),
          tolerance: scoringToleranceSchema.optional(),
          labels: z.record(z.string()).optional(),
        }),
      ),
    }),
  ),
});

export type ParsedBatchConfig = z.infer<typeof batchSchema>;
```

```ts
import type {
  BatchConfig,
  BatchObjectConfig,
  BatchTrialConfig,
  CompiledBatch,
  ObjectScoringConfig,
  ScoringReferenceObject,
} from "../types/batch";
import type { TaskConfig, TaskObjectConfig } from "../types/config";

export function compileBatch(batch: BatchConfig): CompiledBatch {
  const tasks = batch.trials.map((trial) => ({
    file: `tasks/${trial.task_id}.json`,
    config: compileTrial(batch, trial),
  }));

  const scoringReference = {
    schema: "layouttask.scoring-reference.v1" as const,
    experiment_id: batch.experiment_id,
    tasks: Object.fromEntries(
      batch.trials.map((trial) => [
        trial.task_id,
        {
          qid: trial.qid,
          metadata: trial.metadata,
          objects: compileScoringObjects(trial),
        },
      ]),
    ),
  };

  const generatedFiles = [
    "manifest.json",
    ...tasks.map((task) => task.file),
    "scoring/scoring-reference.json",
    "generation-report.json",
  ];

  return {
    manifest: {
      schema: "layouttask.manifest.v1",
      experiment_id: batch.experiment_id,
      title: batch.title,
      config_version: batch.config_version,
      asset_library: batch.shared.asset_library,
      background_library: batch.shared.background_library,
      behavior_library: batch.shared.behavior_library,
      tasks: tasks.map((task) => ({
        qid: task.config.qid,
        task_id: task.config.task_id,
        file: task.file,
      })),
    },
    tasks,
    scoringReference,
    report: {
      schema: "layouttask.generation-report.v1",
      experiment_id: batch.experiment_id,
      task_count: tasks.length,
      generated_files: generatedFiles,
      warnings: [],
    },
  };
}

function compileTrial(batch: BatchConfig, trial: BatchTrialConfig): TaskConfig {
  return {
    schema: "layouttask.task.v1",
    task_id: trial.task_id,
    qid: trial.qid,
    title: trial.title,
    world: trial.world ?? requireShared(batch.shared.world, "world"),
    background: trial.background,
    objects: trial.objects.map(toRuntimeObject),
    completion: trial.completion ?? batch.shared.completion,
    recording: trial.recording ?? batch.shared.recording,
    output: trial.output ?? batch.shared.output,
    data_save: trial.data_save ?? batch.shared.data_save,
    display_image: trial.display_image,
    flow: trial.flow ?? batch.shared.flow,
    stage: trial.stage ?? batch.shared.stage,
    messages: trial.messages ?? batch.shared.messages,
    requirements: trial.requirements ?? batch.shared.requirements,
  };
}

function toRuntimeObject(object: BatchObjectConfig): TaskObjectConfig {
  return {
    id: object.id,
    asset: object.asset,
    x: object.x,
    y: object.y,
    rotation: object.rotation,
    width: object.width,
    height: object.height,
    anchor: object.anchor,
    behavior: object.behavior,
  };
}

function compileScoringObjects(trial: BatchTrialConfig): Record<string, ScoringReferenceObject> {
  const scoringObjects: Record<string, ScoringReferenceObject> = {};

  for (const object of trial.objects) {
    const scoring = mergeObjectScoring(trial.scoring?.objects?.[object.id], object.scoring);
    const target = object.target ?? scoring?.target;
    const role = object.role;

    if (!role && !object.group_id && !target && !scoring?.tolerance && !scoring?.labels) {
      continue;
    }

    scoringObjects[object.id] = {
      role,
      group_id: object.group_id,
      target,
      tolerance: scoring?.tolerance ?? trial.scoring?.default_tolerance,
      labels: scoring?.labels,
    };
  }

  return scoringObjects;
}

function mergeObjectScoring(
  trialScoring?: ObjectScoringConfig,
  objectScoring?: ObjectScoringConfig,
): ObjectScoringConfig | undefined {
  if (!trialScoring) {
    return objectScoring;
  }

  if (!objectScoring) {
    return trialScoring;
  }

  return {
    ...trialScoring,
    ...objectScoring,
    target: objectScoring.target ?? trialScoring.target,
    tolerance: objectScoring.tolerance ?? trialScoring.tolerance,
    labels: { ...trialScoring.labels, ...objectScoring.labels },
  };
}

function requireShared<T>(value: T | undefined, name: string): T {
  if (!value) {
    throw new Error(`Missing shared ${name}`);
```

```ts
#!/usr/bin/env tsx

import { mkdir, readFile, writeFile } from "node:fs/promises";
import path from "node:path";
import { compileBatch } from "../../src/core/batch-compiler";
import { batchSchema } from "../../src/schemas/batch.schema";

async function main(): Promise<void> {
  const input = readArg("--input") ?? process.argv[2];
  const out = readArg("--out") ?? "public/layout-task-generated";

  if (!input) {
    process.stderr.write("Usage: tsx tools/generator/compile-batch.ts --input batch.json --out public/layout-task-generated\n");
    process.exitCode = 1;
    return;
  }

  const text = await readFile(input, "utf8");
  const batch = batchSchema.parse(JSON.parse(text));
  const compiled = compileBatch(batch);

  await mkdir(path.join(out, "tasks"), { recursive: true });
  await mkdir(path.join(out, "scoring"), { recursive: true });

  await writeJson(path.join(out, "manifest.json"), compiled.manifest);
  for (const task of compiled.tasks) {
    await writeJson(path.join(out, task.file), task.config);
  }
  await writeJson(path.join(out, "scoring", "scoring-reference.json"), compiled.scoringReference);
  await writeJson(path.join(out, "generation-report.json"), compiled.report);

  process.stdout.write(`Compiled ${compiled.tasks.length} task(s) to ${out}\n`);
}

async function writeJson(file: string, data: unknown): Promise<void> {
  await mkdir(path.dirname(file), { recursive: true });
  await writeFile(file, `${JSON.stringify(data, null, 2)}\n`, "utf8");
}

function readArg(name: string): string | undefined {
  const index = process.argv.indexOf(name);
  if (index < 0) {
    return undefined;
  }
  return process.argv[index + 1];
}

void main().catch((error) => {
  const message = error instanceof Error ? error.message : "Unknown compile error";
  process.stderr.write(`${message}\n`);
  process.exitCode = 1;
});
```

```ts
    for (const objectId of Object.keys(result.final_state)) {
      const relative = getRelativeState(result, objectId);
      const absolute = getAbsoluteState(result, objectId);
      const reference = scoringReference?.tasks[result.task_id]?.objects[objectId];
      const targetRelative = reference?.target?.relative;
      const targetAbsolute = reference?.target?.absolute;

      rows.push({
        source_index: record.sourceIndex,
        hash_ok: record.decoded.hashOk,
        header_ok: record.decoded.headerOk,
        exp: result.exp,
        qid: result.qid,
        task_id: result.task_id,
        session: result.session,
        object_id: objectId,
        object_role: reference?.role ?? "",
        object_group_id: reference?.group_id ?? "",
        relative_dx_steps: relative?.dx_steps ?? "",
        relative_dy_steps: relative?.dy_steps ?? "",
        relative_rotation_steps: relative?.rotation_steps ?? "",
        target_relative_dx_steps: targetRelative?.dx_steps ?? "",
        target_relative_dy_steps: targetRelative?.dy_steps ?? "",
        target_relative_rotation_steps: targetRelative?.rotation_steps ?? "",
        error_relative_dx_steps: diff(relative?.dx_steps, targetRelative?.dx_steps),
        error_relative_dy_steps: diff(relative?.dy_steps, targetRelative?.dy_steps),
        error_relative_rotation_steps: diff(relative?.rotation_steps, targetRelative?.rotation_steps),
        absolute_x: absolute?.x ?? "",
        absolute_y: absolute?.y ?? "",
        absolute_rotation_deg: absolute?.r ?? "",
        target_absolute_x: targetAbsolute?.x ?? "",
        target_absolute_y: targetAbsolute?.y ?? "",
        target_absolute_rotation_deg: targetAbsolute?.rotation_deg ?? "",
        error_absolute_x: diff(absolute?.x, targetAbsolute?.x),
        error_absolute_y: diff(absolute?.y, targetAbsolute?.y),
        error_absolute_distance: distance(absolute?.x, absolute?.y, targetAbsolute?.x, targetAbsolute?.y),
        error_absolute_rotation_deg: rotationError(absolute?.r, targetAbsolute?.rotation_deg),
      });
    }
  }

  return rows;
}

function getRelativeState(
  result: DecodedSourceRecord["decoded"]["result"],
  objectId: string,
): RelativeFinalObjectState | undefined {
  if (isRelativeFinalState(result.final_state)) {
    return result.final_state[objectId];
  }

  const absolute = result.final_state[objectId] as FinalObjectState | undefined;
  if (absolute?.offsets) {
    return {
      dx_steps: absolute.offsets.xSteps,
      dy_steps: absolute.offsets.ySteps,
      rotation_steps: absolute.offsets.rotationSteps,
    };
  }

  return undefined;
}

function getAbsoluteState(
  result: DecodedSourceRecord["decoded"]["result"],
  objectId: string,
): FinalObjectState | undefined {
  if (isAbsoluteFinalState(result.final_state)) {
    return result.final_state[objectId];
  }

  const relative = result.final_state[objectId] as RelativeFinalObjectState | undefined;
  const context = result.context?.objects[objectId];
  if (!relative || !context) {
    return undefined;
  }

  return {
    id: objectId,
    x: context.origin.x + relative.dx_steps * context.movement_step,
    y: context.origin.y + relative.dy_steps * context.movement_step,
    r: normalizeRotation(context.origin.r + relative.rotation_steps * context.rotation_step),
    counts: { left: 0, right: 0, up: 0, down: 0, cw: 0, ccw: 0 },
    offsets: {
      xSteps: relative.dx_steps,
      ySteps: relative.dy_steps,
      rotationSteps: relative.rotation_steps,
    },
  };
}

function diff(observed?: number, target?: number): number | "" {
  if (observed === undefined || target === undefined) {
    return "";
  }
  return observed - target;
}

function distance(x?: number, y?: number, targetX?: number, targetY?: number): number | "" {
  if (x === undefined || y === undefined || targetX === undefined || targetY === undefined) {
    return "";
  }
  return Math.hypot(x - targetX, y - targetY);
}

function rotationError(rotation?: number, target?: number): number | "" {
  if (rotation === undefined || target === undefined) {
    return "";
  }

  const raw = Math.abs(normalizeRotation(rotation) - normalizeRotation(target));
  return Math.min(raw, 360 - raw);
}

function normalizeRotation(rotation: number): number {
  const normalized = rotation % 360;
  return normalized < 0 ? normalized + 360 : normalized;
}
```

```ts
#!/usr/bin/env tsx

import {
  HelpRequested,
  decodeSourceRecords,
  parseDecoderArgs,
  printUsage,
  readSourceRecords,

    const scoringPath = readArg("--scoring");
    const options = parseDecoderArgs(process.argv.slice(2).filter((arg, index, argv) => {
      return arg !== "--scoring" && argv[index - 1] !== "--scoring";
    }));

    const records = await readSourceRecords(options);
    const decoded = await decodeSourceRecords(records);
    const scoring = scoringPath ? await readScoringReference(scoringPath) : undefined;
    const csv = serializeCsv(toObjectStateRows(decoded, scoring));

      return;

    const message = error instanceof Error ? error.message : "Unknown object export error";
    process.stderr.write(`${message}\n`);
    process.exitCode = 1;
  }
}

function readArg(name: string): string | undefined {
  const index = process.argv.indexOf(name);
  if (index < 0) {
    return undefined;
  }
  return process.argv[index + 1];
```

```ts
import type { LayoutTaskEvent } from "../types/events";
import type { ObjectPose } from "../types/events";
import type { ResultFlowInfo } from "../types/result";

export interface AutosaveKeyParts {
  experimentId: string;
  taskId: string;
  qid: string;
  session: string;
}

export interface LayoutTaskDraft {
  schema: "layouttask.draft.v1";
  saved_at: number;
  restore_count: number;
  state: Record<string, ObjectPose>;
  events: LayoutTaskEvent[];
  flow?: ResultFlowInfo;
}

export interface AutosaveService {
  save(key: string, draft: LayoutTaskDraft): void;
  load(key: string): LayoutTaskDraft | null;
  clear(key: string): void;
}

export function createAutosaveKey(parts: AutosaveKeyParts): string {
  return ["layouttask", "draft", parts.experimentId, parts.taskId, parts.qid, parts.session].join(":");
}

export function createAutosaveService(storage: Storage): AutosaveService {
  return {
    save(key, draft) {
      storage.setItem(key, JSON.stringify(draft));
    },
    load(key) {
      const raw = storage.getItem(key);
      if (!raw) {
        return null;
      }

      try {
        const parsed = JSON.parse(raw) as LayoutTaskDraft;
        if (parsed.schema !== "layouttask.draft.v1") {
          return null;
        }
        return parsed;
      } catch {
        return null;
      }
    },
    clear(key) {
      storage.removeItem(key);
    },
  };
}
```
